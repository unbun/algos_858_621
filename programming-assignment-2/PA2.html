class MNISTFeatureExtraction:
    """MNIST classification using DCT-based feature extraction and eigen decomposition"""
    
    def __init__(self, n_samples_per_class=100, n_eigenvectors=20, verbose=True):
        self.n_samples_per_class = n_samples_per_class
        self.n_eigenvectors = n_eigenvectors
        self.verbose = verbose
        
    def load_data(self):
        if self.verbose:
            print("=" * 60)
            print("PART 1: PREPROCESSING AND VISUALIZATION")
            print("=" * 60)
        
        mnist = fetch_openml('mnist_784', version=1, as_frame=False)
        self.X_full = mnist.data
        self.y_full = mnist.target.astype(int)
        
        if self.verbose:
            print(f"\nDataset Structure:")
            print(f"  Total observations: {len(self.X_full)}")
            print(f"  Features per image: {self.X_full.shape[1]} (flattened 28x28)")
            print(f"  Classes: 0-9 (handwritten digits)")
        
        indices = []
        for digit in range(10):
            digit_indices = np.where(self.y_full == digit)[0]
            sampled = np.random.choice(digit_indices, self.n_samples_per_class, replace=False)
            indices.extend(sampled)
        
        np.random.shuffle(indices)
        self.X = self.X_full[indices]
        self.y = self.y_full[indices]
        
        if self.verbose:
            print(f"\nSampled Dataset:")
            print(f"  Observations: {len(self.X)}")
            print(f"  Samples per class: {self.n_samples_per_class}")
            unique, counts = np.unique(self.y, return_counts=True)
            print(f"  Class distribution: {dict(zip(unique, counts))}")
    
    def visualize_samples(self):
        fig, axes = plt.subplots(2, 5, figsize=(12, 5))
        axes = axes.flatten()
        
        for digit in range(10):
            idx = np.where(self.y == digit)[0][0]
            img = self.X[idx].reshape(28, 28)
            axes[digit].imshow(img, cmap='gray')
            axes[digit].set_title(f'Digit: {digit}')
            axes[digit].axis('off')
        
        plt.suptitle('Example Handwritten Digits from MNIST', fontsize=14)
        plt.tight_layout()
        plt.show()
    
    def create_dct_masks(self, size=28):
        h_mask = np.zeros((size, size), dtype=bool)
        v_mask = np.zeros((size, size), dtype=bool)
        d_mask = np.zeros((size, size), dtype=bool)

        for i in range(size):
            for j in range(size):
                if i >= j:
                    h_mask[i, j] = True
                if j >= i:
                    v_mask[i, j] = True
                if abs(i - j) <= 1:
                    d_mask[i, j] = True

        return h_mask, v_mask, d_mask
    
    def apply_2d_dct(self, image):
        img_2d = image.reshape(28, 28)
        dct_row = dct(img_2d, type=2, norm='ortho', axis=0)
        dct_2d = dct(dct_row, type=2, norm='ortho', axis=1)
        return dct_2d
    
    def extract_dct_features(self):
        if self.verbose:
            print("\n" + "=" * 60)
            print("PART 2: FEATURE ENGINEERING (DCT + Eigen Decomposition)")
            print("=" * 60)
        
        h_mask, v_mask, d_mask = self.create_dct_masks()
        
        h_coeffs, v_coeffs, d_coeffs = [], [], []
        for img in self.X:
            dct_2d = self.apply_2d_dct(img)
            h_coeffs.append(dct_2d[h_mask].flatten())
            v_coeffs.append(dct_2d[v_mask].flatten())
            d_coeffs.append(dct_2d[d_mask].flatten())
        
        h_matrix = np.array(h_coeffs)
        v_matrix = np.array(v_coeffs)
        d_matrix = np.array(d_coeffs)
        
        if self.verbose:
            print(f"\nDCT Coefficient Matrices:")
            print(f"  Horizontal: {h_matrix.shape}")
            print(f"  Vertical: {v_matrix.shape}")
            print(f"  Diagonal: {d_matrix.shape}")
        
        def compute_eigen_features(data, name):
            data_centered = data - data.mean(axis=0)
            cov = np.cov(data_centered.T)
            eigenvalues, eigenvectors = np.linalg.eigh(cov)
            idx = np.argsort(eigenvalues)[::-1]
            eigenvectors = eigenvectors[:, idx]
            top_eigenvectors = eigenvectors[:, :self.n_eigenvectors]
            projected = data_centered @ top_eigenvectors
            variance_explained = eigenvalues[idx][:self.n_eigenvectors].sum() / eigenvalues.sum()
            if self.verbose:
                print(f"  {name}: Top {self.n_eigenvectors} eigenvectors explain {variance_explained*100:.1f}% variance")
            return projected
        
        print("\nEigen Decomposition:")
        h_features = compute_eigen_features(h_matrix, "Horizontal")
        v_features = compute_eigen_features(v_matrix, "Vertical")
        d_features = compute_eigen_features(d_matrix, "Diagonal")
        
        self.X_features = np.hstack([h_features, v_features, d_features])
        
        if self.verbose:
            print(f"\nFinal Feature Matrix: {self.X_features.shape}")
            print(f"  (60 features = 20 horizontal + 20 vertical + 20 diagonal)")
    
    def prepare_data(self, test_size=0.2):
        self.X_train, self.X_test, self.y_train, self.y_test = train_test_split(
            self.X_features, self.y, test_size=test_size, random_state=42, stratify=self.y
        )
        self.X_train_raw, self.X_test_raw, _, _ = train_test_split(
            self.X, self.y, test_size=test_size, random_state=42, stratify=self.y
        )
        
        self.scaler = StandardScaler()
        self.X_train_scaled = self.scaler.fit_transform(self.X_train)
        self.X_test_scaled = self.scaler.transform(self.X_test)
        
        if self.verbose:
            print(f"\nTrain/Test Split: Training={len(self.X_train)}, Test={len(self.X_test)}")
    
    def train_sklearn_classifier(self):
        if self.verbose:
            print("\n" + "=" * 60)
            print("PART 3: CLASSIFICATION ALGORITHMS")
            print("=" * 60)
            print("\n--- Model 1: Random Forest (sklearn) ---")
        
        start = time.time()
        self.rf_model = RandomForestClassifier(n_estimators=100, random_state=42, n_jobs=-1)
        self.rf_model.fit(self.X_train_scaled, self.y_train)
        rf_time = time.time() - start
        
        train_pred = self.rf_model.predict(self.X_train_scaled)
        test_pred = self.rf_model.predict(self.X_test_scaled)
        
        self.rf_train_acc = accuracy_score(self.y_train, train_pred)
        self.rf_test_acc = accuracy_score(self.y_test, test_pred)
        self.rf_pred = test_pred
        
        if self.verbose:
            print(f"Training Accuracy: {self.rf_train_acc*100:.2f}%")
            print(f"Test Accuracy: {self.rf_test_acc*100:.2f}%")
            print(f"Training Time: {rf_time:.2f}s")
            
            importances = self.rf_model.feature_importances_
            print(f"\nTop 5 Important Features:")
            top_idx = np.argsort(importances)[::-1][:5]
            for i, idx in enumerate(top_idx):
                direction = "Horizontal" if idx < 20 else ("Vertical" if idx < 40 else "Diagonal")
                local_idx = idx % 20
                print(f"  {i+1}. Feature {idx} ({direction} #{local_idx}): {importances[idx]:.4f}")
    
    def train_custom_svm(self):
        if self.verbose:
            print("\n--- Model 2: Custom SVM (One-vs-Rest) ---")
        
        n_train = min(500, len(self.X_train_scaled))
        X_svm = self.X_train_scaled[:n_train]
        y_svm = self.y_train[:n_train]
        
        start = time.time()
        
        def train_binary_svm(X, y_binary, C=1.0):
            n = len(X)
            y = y_binary.astype(float)
            K = X @ X.T
            P = matrix(np.outer(y, y) * K)
            q = matrix(-np.ones(n))
            G = matrix(np.vstack([-np.eye(n), np.eye(n)]))
            h = matrix(np.hstack([np.zeros(n), C * np.ones(n)]))
            A = matrix(y.reshape(1, -1))
            b = matrix(np.zeros(1))
            sol = solvers.qp(P, q, G, h, A, b)
            alpha = np.array(sol['x']).flatten()
            sv_mask = alpha > 1e-5
            w = (alpha * y) @ X
            sv_idx = np.where(sv_mask)[0]
            b_val = np.mean(y[sv_idx] - X[sv_idx] @ w) if len(sv_idx) > 0 else 0
            return w, b_val
        
        self.svm_models = {}
        for digit in range(10):
            y_binary = np.where(y_svm == digit, 1, -1)
            w, b = train_binary_svm(X_svm, y_binary)
            self.svm_models[digit] = (w, b)
        
        svm_time = time.time() - start
        
        def predict_svm(X):
            scores = np.zeros((len(X), 10))
            for digit, (w, b) in self.svm_models.items():
                scores[:, digit] = X @ w + b
            return np.argmax(scores, axis=1)
        
        train_pred = predict_svm(X_svm)
        test_pred = predict_svm(self.X_test_scaled)
        
        self.svm_train_acc = accuracy_score(y_svm, train_pred)
        self.svm_test_acc = accuracy_score(self.y_test, test_pred)
        self.svm_pred = test_pred
        
        if self.verbose:
            print(f"Training Accuracy: {self.svm_train_acc*100:.2f}%")
            print(f"Test Accuracy: {self.svm_test_acc*100:.2f}%")
            print(f"Training Time: {svm_time:.2f}s (on {n_train} samples)")
    
    def train_cnn(self, epochs=10):
        if self.verbose:
            print("\n--- Model 3: Convolutional Neural Network ---")
        
        X_train_t = torch.FloatTensor(self.X_train_raw.reshape(-1, 1, 28, 28) / 255.0)
        X_test_t = torch.FloatTensor(self.X_test_raw.reshape(-1, 1, 28, 28) / 255.0)
        y_train_t = torch.LongTensor(self.y_train)
        y_test_t = torch.LongTensor(self.y_test)
        
        train_loader = DataLoader(TensorDataset(X_train_t, y_train_t), batch_size=64, shuffle=True)
        
        class SimpleCNN(nn.Module):
            def __init__(self):
                super().__init__()
                self.conv1 = nn.Conv2d(1, 32, 3, padding=1)
                self.conv2 = nn.Conv2d(32, 64, 3, padding=1)
                self.pool = nn.MaxPool2d(2, 2)
                self.fc1 = nn.Linear(64 * 7 * 7, 128)
                self.fc2 = nn.Linear(128, 10)
                self.relu = nn.ReLU()
                self.dropout = nn.Dropout(0.25)
            
            def forward(self, x):
                x = self.pool(self.relu(self.conv1(x)))
                x = self.pool(self.relu(self.conv2(x)))
                x = x.view(-1, 64 * 7 * 7)
                x = self.dropout(self.relu(self.fc1(x)))
                return self.fc2(x)
        
        model = SimpleCNN()
        criterion = nn.CrossEntropyLoss()
        optimizer = optim.Adam(model.parameters(), lr=0.001)
        
        start = time.time()
        for epoch in range(epochs):
            model.train()
            for Xb, yb in train_loader:
                optimizer.zero_grad()
                loss = criterion(model(Xb), yb)
                loss.backward()
                optimizer.step()
        cnn_time = time.time() - start
        
        model.eval()
        with torch.no_grad():
            train_pred = torch.argmax(model(X_train_t), dim=1).numpy()
            test_pred = torch.argmax(model(X_test_t), dim=1).numpy()
        
        self.cnn_train_acc = accuracy_score(self.y_train, train_pred)
        self.cnn_test_acc = accuracy_score(self.y_test, test_pred)
        self.cnn_pred = test_pred
        
        if self.verbose:
            print(f"Training Accuracy: {self.cnn_train_acc*100:.2f}%")
            print(f"Test Accuracy: {self.cnn_test_acc*100:.2f}%")
            print(f"Training Time: {cnn_time:.2f}s ({epochs} epochs)")
    
    def compare_models(self):
        if self.verbose:
            print("\n" + "=" * 60)
            print("MODEL COMPARISON")

        print(f"{'CNN':<20} {self.cnn_train_acc*100:<12.1f} {self.cnn_test_acc*100:<12.1f}")
        print("-" * 50)
    
    def pipeline(self):

Reflection¶

Part 1 Analysis: The MNIST dataset contains 70,000 grayscale images of handwritten digits (0-9), each stored as a flattened 784-feature vector (28x28 pixels). We sampled 100 images per class for computational efficiency while maintaining balanced class representation.

Part 2 Analysis: The 2D DCT transforms spatial pixel intensities into frequency domain coefficients. The directional masks capture: horizontal structures (upper triangle, 406 coefficients), vertical structures (lower triangle, 406 coefficients), and diagonal patterns (near-diagonal band, 82 coefficients). Eigen decomposition on each direction's coefficients identifies the 20 most informative orthogonal components per direction, achieving 75-93% variance explained while reducing dimensionality from hundreds of coefficients to just 60 features total.

Part 3 Analysis:

  • Random Forest achieved 100% training / 88% test accuracy on DCT features. Tree-based methods handle non-linear boundaries effectively; the feature importance shows vertical and diagonal components contribute most (Vertical #0: 6.7%, Diagonal #3: 4.7%, Horizontal #0: 4.8%).
  • Custom SVM (One-vs-Rest with linear kernel) achieved 96% training / 81% test accuracy. The linear kernel is limited by decision boundary constraints, and the 500-sample training subset reduced its capacity.
  • CNN achieved 97% training / 91.5% test accuracy on raw pixels—the highest test performance despite not using engineered features.

Why CNN outperforms: CNNs learn hierarchical spatial features end-to-end through convolutional filters, automatically discovering task-specific representations (edges, curves, strokes). The CNN's inductive bias—translation invariance and local connectivity—is well-suited for image data. DCT features are effective but may discard discriminative spatial details that convolutions preserve through their hierarchical structure.

def haversine(lat1, lon1, lat2, lon2):
    """Calculate the great-circle distance between two points (in meters)."""
    lat1_rad, lon1_rad = math.radians(lat1), math.radians(lon1)
    lat2_rad, lon2_rad = math.radians(lat2), math.radians(lon2)
    dlat = lat2_rad - lat1_rad
    dlon = lon2_rad - lon1_rad
    a = math.sin(dlat/2)**2 + math.cos(lat1_rad) * math.cos(lat2_rad) * math.sin(dlon/2)**2
    c = 2 * math.asin(math.sqrt(a))
    return R * c

def update_position(lat, lon, speed, heading, dt):
    """
    Update position based on current lat/lon, speed (m/s), heading (degrees),
    and time step dt. Uses a simple spherical approximation.
    """
    distance = speed * dt
    heading_rad = math.radians(heading)
    delta_north = distance * math.cos(heading_rad)
    delta_east = distance * math.sin(heading_rad)
    delta_lat = (delta_north / R) * (180 / math.pi)
    delta_lon = (delta_east / (R * math.cos(math.radians(lat)))) * (180 / math.pi)
    new_lat = lat + delta_lat
    new_lon = lon + delta_lon
    new_lon = (new_lon + 180) % 360 - 180  # normalize longitude
    new_lat = max(min(new_lat, 90), -90)   # constrain latitude
    return new_lat, new_lon

def horizon_distance(alt):
    """
    Calculate the horizon distance (in meters) for a given altitude 'alt'
    using the approximation: distance ≈ √(2 * R * alt)
    """
    return math.sqrt(2 * R * alt)

def simulate_run():
    # Choose a random ground object location within the ground box.
    ground_lon = random.uniform(ground_lon_min, ground_lon_max)
    ground_lat = random.uniform(ground_lat_min, ground_lat_max)
    ground_alt = 1.5  # observer height in meters

    init_plane_lat = random.uniform(aircraft_lat_min, aircraft_lat_max)
    init_plane_lon = random.uniform(aircraft_lon_min, aircraft_lon_max)

    plane_speed = 250  # m/s (~900 km/h)
    plane_heading = random.uniform(0, 360)  # degrees

    # Set initial aircraft position.
    plane_lat = init_plane_lat
    plane_lon = init_plane_lon

    # Flag for line-of-sight occurrence.
    los_occurred = False

    for step in range(int(num_steps) + 1):
        # Calculate the great-circle distance between ground object and aircraft.
        distance = haversine(ground_lat, ground_lon, plane_lat, plane_lon)
        # Calculate horizon distances.
        d_ground = horizon_distance(ground_alt)
        d_plane = horizon_distance(plane_alt)
        # If LOS exists at this timestep, flag it.
        if distance <= (d_ground + d_plane):
            los_occurred = True
            break
        # Update aircraft position.
        plane_lat, plane_lon = update_position(plane_lat, plane_lon, plane_speed, plane_heading, dt)

    # If LOS occurred at least once, target is 0, otherwise 1.
    target = 0 if los_occurred else 1

        "ground_lon": ground_lon,
        "ground_alt": ground_alt,
        "init_plane_lat": init_plane_lat,
        "init_plane_lon": init_plane_lon,
        "plane_alt": plane_alt,
        "plane_speed": plane_speed,
        "plane_heading": plane_heading,
        "target": target

Simulation Overview: The simulation models line-of-sight (LOS) detection between a ground observer and an aircraft. It defines two geographic bounding boxes: a smaller "ground object" box (Florida region, ~2° longitude × 1.5° latitude) where ground observers are randomly placed, and a larger "aircraft" box (~7× wider, centered on the same location) where aircraft start their flight paths.

LOS Detection Mechanism: At each 10-second timestep over a 1-hour flight, the code computes the great-circle (haversine) distance between ground and aircraft positions. LOS exists when this distance is less than or equal to the sum of two horizon distances—calculated using the approximation √(2Rh) where R is Earth's radius (~6371 km) and h is altitude. This formula represents how far each observer can "see" given their elevation above Earth's curved surface: higher altitude = longer horizon distance = greater detection range.

Target Variable:

  • target = 0 means LOS occurred at least once during the flight (detection success)
  • target = 1 means no LOS ever occurred throughout the entire simulation (detection failure)

MODEL 1: GPT-2 (Transformer - Autoregressive Text Generation)
======================================================================

--- (a) Model Demonstration ---
Input: "Machine learning is transforming"
Output: "Machine learning is transforming the way we interact with
         technology. From voice assistants to recommendation systems,
         AI is becoming an integral part of our daily lives..."

Input/Output Representation:
  - Input: Text string -> tokenized to integer IDs -> embeddings (768-dim)
  - Output: Probability distribution over vocabulary -> sampled tokens -> decoded text

--- (b) Architecture ---
GPT-2 Architecture (Decoder-only Transformer):
  1. Token Embedding: Maps vocab IDs to 768-dim vectors
  2. Positional Embedding: Learned position encodings (max 1024 positions)
  3. Transformer Decoder Blocks (12 layers):
     - Masked Multi-Head Self-Attention (12 heads, 64 dim each)
     - Layer Normalization
     - Feed-Forward Network (MLP: 768 -> 3072 -> 768)
     - Residual Connections
  4. Output Layer: Projects to vocab size (50,257 tokens)

Model Parameters: 124,439,808 (GPT-2 small)

--- (c) Loss Function ---
Cross-Entropy Loss (Language Modeling):
  L = -sum_{t=1}^{T} log P(token_t | token_1, ..., token_{t-1})

  - Autoregressive: Predicts next token given all previous tokens
  - Teacher forcing: Uses ground truth during training
  - Softmax converts logits to probabilities over discrete vocabulary

--- (d) Training Challenge & Mitigation ---
Challenge: Exposure Bias
  - During training: model sees ground truth previous tokens
  - During inference: model sees its own (potentially wrong) predictions
  - Errors can accumulate and compound during generation

Mitigation Strategies:
  1. Large-scale pretraining: GPT-2 trained on 8M web pages (WebText)
  2. Sampling techniques: Temperature, nucleus (top-p), top-k sampling
  3. Data diversity: Massive corpus reduces overfitting to specific patterns

MODEL 2: T5 (Transformer - Text-to-Text Summarization)
======================================================================

--- (a) Model Demonstration ---
Input: "summarize: Machine learning is a subset of artificial intelligence 
that enables..."
Output: "Machine learning enables systems to learn from data without explicit programming."

Input/Output Representation:
  - Input: Text with task prefix ('summarize:') -> tokenized -> embeddings
  - Output: Compressed summary tokens -> decoded to text

--- (b) Architecture ---
T5 Architecture (Encoder-Decoder Transformer):
  ENCODER (6 layers):
    1. Token + Relative Position Embeddings
    2. Encoder Blocks:
       - Bidirectional Self-Attention (can see full input)
       - Layer Normalization
       - Feed-Forward Network
  DECODER (6 layers):
    1. Token + Relative Position Embeddings
    2. Decoder Blocks:
       - Masked Self-Attention (causal, left-to-right)
       - Cross-Attention (attends to encoder outputs)
       - Feed-Forward Network
    3. Output projection to vocabulary

Model Parameters: 60,506,624 (T5-small)

--- (c) Loss Function ---
Cross-Entropy Loss (Seq2Seq):
  L = -sum_{t=1}^{T} log P(y_t | y_1,...,y_{t-1}, X)

  - Encoder processes full input X bidirectionally
  - Decoder autoregressively generates output Y
  - Cross-attention allows decoder to 'look at' relevant input parts

--- (d) Training Challenge & Mitigation ---
Challenge: Training Instability at Scale
  - Deep transformers prone to vanishing/exploding gradients
  - Large vocabulary and long sequences increase memory/compute
  - Multi-task learning can cause task interference

Mitigation Strategies:
  1. Relative position encodings: Better generalization to longer sequences
  2. Pre-LayerNorm: More stable gradients in deep networks
  3. Multi-task pretraining: C4 corpus (750GB), unified text-to-text format

Reflection¶

Model 1: GPT-2 (Transformer - Text Generation)

(a) Demonstration: GPT-2 generates coherent text continuations from a prompt. Input text is tokenized into integer IDs using BPE (Byte-Pair Encoding), embedded into 768-dimensional vectors, and processed through transformer layers. Output is a probability distribution over 50,257 vocabulary tokens, sampled autoregressively.

(b) Architecture: Decoder-only transformer with 12 layers. Each layer contains masked multi-head self-attention (12 heads), layer normalization, and a feed-forward MLP. "Masked" means each token can only attend to previous tokens, enforcing causal (left-to-right) generation.

(c) Loss Function: Cross-entropy over next-token predictions: L = -Σ log P(token_t | previous tokens). Teacher forcing provides ground truth during training; softmax converts logits to probabilities over the discrete vocabulary.

(d) Challenge & Mitigation: Exposure bias—model sees ground truth during training but its own (possibly wrong) outputs at inference. Mitigated via large-scale pretraining on diverse data (8M web pages), plus sampling techniques (temperature, nucleus/top-p) that improve generation diversity.

Model 2: T5 (Transformer - Summarization)

(a) Demonstration: T5 compresses input text into a summary. Uses task prefixes ("summarize:") to unify different NLP tasks under text-to-text format. Input/output both represented as token sequences.

(b) Architecture: Encoder-decoder transformer. Encoder (6 layers) uses bidirectional self-attention to understand full input. Decoder (6 layers) uses masked self-attention plus cross-attention to encoder outputs, generating one output token at a time.

(c) Loss Function: Sequence-to-sequence cross-entropy: L = -Σ log P(y_t | y_{<t}, X). Cross-attention mechanism allows decoder to dynamically "focus" on relevant encoder states for each output step.

(d) Challenge & Mitigation: Training instability with deep networks and long sequences. Mitigated via relative position encodings (better length generalization), layer normalization, and massive multi-task pretraining on C4 corpus (750GB).

Comparison Paragraph:

GPT-2's key strength is its simplicity and generation fluency—as a decoder-only model, it excels at open-ended text generation by predicting one token at a time autoregressively. T5's strength lies in its encoder-decoder architecture with cross-attention, making it superior for tasks requiring input understanding before generation (summarization, translation, Q&A).

GPT-2's limitation is its lack of bidirectional context—it can only attend to previous tokens, missing future context that might improve coherence. T5's limitation is higher computational cost due to its dual-stack architecture and cross-attention overhead, requiring more memory and time for inference.

In practice, the trade-off centers on task requirements: for creative generation or chatbots where context flows forward, GPT-2's efficiency wins. For transformation tasks (summarization, translation) where understanding the full input before responding is critical, T5's comprehension-then-generation paradigm is preferred despite the compute penalty.
